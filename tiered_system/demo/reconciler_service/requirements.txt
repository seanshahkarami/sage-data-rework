# The reconciler itself only reads sqlite (stdlib) and makes one
# authenticated GET + PUT. It needs pyiceberg anyway: it reuses
# pelican_fileio for the authenticated transport, and that module imports
# `pyiceberg.io.FileIO` at module scope as PelicanFileIO's base class --
# a base class is needed at class-definition time, so the import cannot be
# deferred without restructuring a file the experiments already test.
#
# Reusing the tested transport is worth the image size. Duplicating the
# SciToken auth path here to save ~370 MB on a local demo would put a
# second, untested copy of the credential code in the tree.
pyiceberg==0.12.0
PyJWT[crypto]==2.10.1
requests==2.34.2
